class Test:
    def get(self, x, y, z):
        self.x = x
        self.y = y
        self.z = z

    def add(self, m, n):
        self.x = m.x + n.x
        self.y = m.y + n.y
        self.z = m.z + n.z

    def disp(self):
        print("x =", self.x, ", y =", self.y, ", z =", self.z)

a = Test()
b = Test()
c = Test()

a.get(10, 20, 30)
b.get(40, 50, 60)
c.add(a, b)

print("First Object:")
a.disp()
print("Second Object:")
b.disp()
print("Third Object (Sum):")
c.disp()
