3) list = ['google.com' , 'rediff.com' , 'gmail.com' , 'amazon.com' , 'netflix.com']

import time

urls =  ['google.com' , 'rediff.com' , 'gmail.com' , 'amazon.com' , 'netflix.com']

    url = random.choice(urls)
