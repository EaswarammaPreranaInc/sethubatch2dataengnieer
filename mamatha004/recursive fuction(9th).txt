def outer():
    print('Outer  function')       #  outer function
    def inner1():
        print('1st  inner  function')
    def inner2():
        print('2nd  inner  function')
    print('Hi')                    # hi
    inner2()                       # calls inner2 
    print('Hello')                 #  Hello
    inner1()                       # calls inner1
    print('Back  to  outer  function')  # back to outer function

# End of the function

#fibbonaci series

def fib(i):
    if i==0:
        return 0
    if  i==1:
        return 1
    else:
       return fib(i-1)+fib(i-2)
        
i=int(input("how many terms: "))
print("fibonacci: ",fib(i))



#power function
def power(a,b):
    if a==0 or b==0:
        return 1
    if  a!=0 and b!=0:
        return a**b
        
a=int(input("enter 1st number: "))
b=int(input("enter 2nd number: "))
print("power : ",power(a,b))
