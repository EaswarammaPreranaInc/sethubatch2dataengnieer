def separate_w(string):
    word=""
    for ch in string:
        if ch !=" ":
            word+=ch
        else:
            if word:
             yield word
             word=""
    if word:
          yield word
string=input("enter any sentence:")
print("string in words")
for w in separate_w(string):
    print(w)

def f1():
    print('One')      # One
    yield 1           # yields 1
    print('Two')      # Two
    yield 2           # yields 2
    print('Three')    # Three
    yield 3           # yields 3
    print('End')      # End
# End of generator

g = f1()
for m in g:
    print(m)          # 1, 2, 3

x, y, z = f1()        #  One, Two, Three, End
print(x)              # 1
print(y)              # 2
print(z)              # 3
