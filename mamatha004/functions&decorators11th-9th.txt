def decor(fun):
    def inner(x, y):
        try:
            return fun(x, y)
        except:
            return 'Division   by  0  is  not  permitted'
    return inner

@decor
def div(a, b):
    return a / b
# End  of  the  function

def decor(fun):
    def inner():
        print(f'Decorating {fun._name_} function')
        fun()
        print('Decoration is finished')
    return inner

@decor
def f1():
    print('Hello')
# End  of the function

f1()                      #  Decorating f1 function
                          #         Hello
                          #         Decoration is finished
