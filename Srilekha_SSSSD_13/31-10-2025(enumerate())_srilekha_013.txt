# How  to  iterate  zip  object  in  differenet  ways  (Home  work)

import time

# Define state and capital lists
states = ['Telangana', 'Andhra Pradesh', 'Karnataka', 'Tamilnadu']
capitals = ['Hyderabad', 'Amaravathi', 'Bangalore', 'Chennai']

# Create zip object
z1 = zip(states, capitals)

# Check type and print zip object reference
print("Type of zip object:", type(z1))
print("Zip object reference:", z1)

# Use next() to get first pair
print("First pair using next():", next(z1))

# Use __next__() method to get second pair
print("Second pair using __next__():", z1.__next__())

# Iterate through zip object with for loop
print("Iterate through zip object with for loop:")
for state, capital in zip(states, capitals):
    print(f"{state:<16} ... {capital}")
    time.sleep(1)

# Unpack zip object using * operator
print("Unpacked zip object with * operator:", list(zip(*zip(states, capitals))))

# Convert zip object to list
print("Zip object as list:", list(zip(states, capitals)))

# Convert zip object to dictionary
print("Zip object as dictionary:", dict(zip(states, capitals)))
