'''


import random
import webbrowser
lst = ['google.com' , 'rediff.com' , 'gmail.com' , 'amazon.com' , 'netflix.com']
import time
while True:
  for file in lst:
   
    webbrowser.open(f"https://{file}")
    sec = random.randint(5,20)
    time.sleep(sec)

import random

choices = ["Rock", "Paper", "Scissors"]

while True:
    user = int(input("What do you want to select (0 - Rock , 1 - Paper , 2 - Scissors)  : "))
    print("User  :  ", choices[user])

    comp = random.randint(0, 2)
    print("Computer  :  ", choices[comp])
    if user == comp:
        print("Draw")
    elif (comp == 1 and user == 0) or (comp == 2 and user == 1) or (comp == 0 and user == 2):
        print("Computer wins")
    else:
        print("User wins")

    ch = input("Continue  ( y / n ) ?  ").lower()
    if ch != 'y':
        print("End of the game")
        break
