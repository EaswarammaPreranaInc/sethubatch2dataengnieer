from threading import Thread, Lock
import time

def f1():
    l1.acquire()
    try:
        time.sleep(1)
        l2.acquire()
        try:
            print('1st thread is under execution')
        finally:
            l2.release()
    finally:
        l1.release()
    print('End of the 1st thread')

def f2():
    l1.acquire()
    try:
        time.sleep(1)
        l2.acquire()
        try:
            print('2nd thread is under execution')
        finally:
            l2.release()
    finally:
        l1.release()
    print('End of the 2nd thread')

l1 = Lock()
l2 = Lock()

t1 = Thread(target=f1)
t2 = Thread(target=f2)
t1.start()
t2.start()
t1.join()
t2.join()

print('End of main thread')
