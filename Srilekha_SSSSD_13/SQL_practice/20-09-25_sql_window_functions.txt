#Retrieve all employees along with their department names.
select e.emp_name, d.dept_name from Employees as e 
join Departments as d on e.dept_id = d.dept_id;
#--------------------------------------------------------------------------------
#List all employees and the projects they are working on (show NULL if no project).
select e.emp_name ,p.project_name from employees as e
left join projects as p on e.dept_id = p.dept_id; 


#--------------------------------------------------------------------------------------

#Show all departments and employees (include departments with no employees).
select emp_name , dept_name from departments  as d
left join employees as e on  d.dept_id = e.dept_id;


#--------------------------------------------------------------------------------------
#Show all projects and employees working on them (include projects without employees).
select emp_name , dept_name from    employees as e  left join  departments  as d
 on  d.dept_id = e.dept_id;

#-----------------------------------------------------------------------

#Retrieve employees along with both department and project details.
select  e.emp_name ,d.dept_name , p.project_name
from employees as e
join departments as d on d.dept_id = e.dept_id
join projects as p on d.dept_id = p.dept_id;

#--------------------------------------------------------------------------------------------
#Find employees who are working on projects belonging to a different department than their own.

SELECT 
    e.emp_name,
    e.dept_id AS employee_dept,
    p.project_name,
    p.dept_id AS project_dept
FROM 
    employees e
JOIN 
    projects p ON e.project_id = p.project_id
WHERE 
    e.dept_id <> p.dept_id;
    
    #----------------------------------------------------------------------
    
   # List all employees along with their manager’s name (self-join).
   select m1.emp_name as employee, 
   m2.emp_name as manager 
   from employees as m1
   left join employees as m2
   on   m1.manager_id =  m2.emp_id ;

#   Show the number of employees working in each department.

SELECT 
    d.dept_name,
    COUNT(e.emp_id) AS employee_count
FROM  departments AS d
LEFT JOIN  employees AS e ON d.dept_id = e.dept_id
GROUP BY  d.dept_name;

#--------------------------------------------------------------------------------

#--------------------------------------------------------------------------------
#  Window functions
#For each employee’s sales, show the previous sale amount (using LAG).

SELECT 
    emp_id,
    sale_date,
    sale_amount,
    LAG(sale_amount) OVER (
        PARTITION BY emp_id 
        ORDER BY sale_date
    ) AS previous_sale
FROM 
    sales;
#-----------------------------------------------------------------------------------

#For each employee’s sales, show the next sale amount (using LEAD).

SELECT 
    emp_id,
    sale_date,
    sale_amount,
    LEAD(sale_amount) OVER (
        PARTITION BY emp_id 
        ORDER BY sale_date
    ) AS next_sale
FROM 
    sales;
    
#Show the difference between each employee’s current sale and their previous sale.
SELECT 
    emp_id,
    sale_date,
    sale_amount,
    LAG(sale_amount) OVER (
        PARTITION BY emp_id 
        ORDER BY sale_date
    ) AS previous_sale,
    sale_amount - LAG(sale_amount) OVER (
        PARTITION BY emp_id 
        ORDER BY sale_date
    ) AS sale_difference
FROM 
    sales;

#-------------------------------------------------------------------------------------
#Assign a row number to each sale of every employee ordered by sale_date.

SELECT 
    emp_id,
    sale_date,
    sale_amount,
    ROW_NUMBER() OVER (
        PARTITION BY emp_id 
        ORDER BY sale_date
    ) AS sale_rank
FROM 
    sales;


#---------------------------------------------------------------------------------
#Find the highest sale made by each employee using a window function (not GROUP BY).

SELECT 
    emp_id,
    sale_date,
    sale_amount,
    MAX(sale_amount) OVER (
        PARTITION BY emp_id
    ) AS highest_sale
FROM 
    sales;
