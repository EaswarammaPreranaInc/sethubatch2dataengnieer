choices=['Rock','Paper','Scissors']
while True:
	n=int(input('What  do  you  want  to  select  (0 - Rock , 1 - Paper , 2 - Scissors)  :'))
	user=choices[n]
	computer=choices[random.randint(0,2)]
	if user==computer:
		print('Draw')
	elif (computer=='Paper' and user=='Rock') or (computer == 'Scissors' and user == 'Paper') or (computer == 'Rock' and user == 'Scissors'):
		print('Computer wins')
	else:
		print('User wins')
	c=input("continue (y/n)? ").lower()
	if c!='y':
		print('End of the game')
		break
