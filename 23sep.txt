# -----------------------------------------------------
# 3. Method overriding in same class
# -----------------------------------------------------
def m1():
    print('Function')

class c1:
    def m1(self):
        print('1st method')
    def m1(self):
        print('2nd method')
    def m1(self):
        print('3rd method')

a = c1()
a.m1()   # Only last definition kept → "3rd method"
m1()     # Calls outside function → "Function"


# -----------------------------------------------------
# 4. Method overloading (without defaults) – Python does not support it
# -----------------------------------------------------
class c1:
    def m1(self):
        print('No argument method')
    def m1(self, x):
        print('Single argument method:', x)
    def m1(self, x, y):
        print('Two argument method:', x, y)

a = c1()
a.m1(10, 20)   # "Two argument method: 10 20"
# a.m1(30)     # ❌ TypeError: missing 1 required positional argument
# a.m1()       # ❌ TypeError: missing 2 required positional arguments


# -----------------------------------------------------
# 5. Method overloading with default values (works in Python)
# -----------------------------------------------------
class c1:
    def m1(self, x=1, y=2):
        print('Two argument method:', x, y)

a = c1()
a.m1(10, 20)   # Two argument method: 10 20
a.m1(30)       # Two argument method: 30 2
a.m1()         # Two argument method: 1 2

# 9. Triangle program
# -----------------------------------------------------
import math

class Triangle:
    def __init__(self, a, b, c):
        self.a = a
        self.b = b
        self.c = c

    def is_valid(self):
        # Triangle inequality check
        return (self.a + self.b > self.c) and \
               (self.a + self.c > self.b) and \
               (self.b + self.c > self.a)

    def area(self):
        s = (self.a + self.b + self.c) / 2
        return math.sqrt(s * (s - self.a) * (s - self.b) * (s - self.c))

    def perimeter(self):
        return self.a + self.b + self.c


# ----------- MAIN EXECUTION -------------
a, b, c = map(float, input("Enter 3 sides of triangle: ").split())
t = Triangle(a, b, c)

if t.is_valid():
    print("Area:", t.area())
    print("Perimeter:", t.perimeter())
else:
    print("Not a triangle")
