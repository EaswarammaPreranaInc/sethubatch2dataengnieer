# add()  method  demo  program  (Home  work)
a = set()
a.add(True)       # {True}
a.add(25)         # {True, 25}
a.add(10.8)       # {True, 25, 10.8}
a.add(1)          # no effect (1 == True already exists)
a.add('Hyd')      # {'Hyd', True, 25, 10.8}
a.add(25)         # duplicate → ignored
a.add(None)       # {None, 'Hyd', True, 25, 10.8}
a.add('Hyd')      # duplicate → ignored
a.add(1.0)        # duplicate (1.0 == True) → ignored
print(a)          # {None, 'Hyd', True, 25, 10.8}
a.add(10,20,30)   # ❌ TypeError: set.add() takes exactly one argument
a.add([10,20,30]) # ❌ TypeError: unhashable type: 'list'


  
# Find  outputs  (Home  work)
a = {25 , 10.8 , 'Hyd' , True}
tpl = (10 , 20 , 30)
print(a)          # {'Hyd', True, 25, 10.8}
print(id(a))      # some unique id 
a.add(tpl)        
a.add('Sec')     
print(a)          # {'Hyd', True, 25, 10.8, (10,20,30), 'Sec'}
print(id(a))      # same id as before (mutable set)
print(len(a))     # 6
a.add([100 , 200 , 300])  # Error
a.add(set())              # Error
a.add({})                 #Error

# copy()  method  demo  program  (Home  work)
a = {10 , 20 , 15 , 18}
print(a)          # {10, 20, 15, 18}
b = a.copy()
print(b)          # {10, 20, 15, 18}
print(a is b)     # False
print(a == b)     # True
c = a
print(a is c)     # True

# remove()  method  demo  program  (Home  work)
a = {25 , 10.8 , 'Hyd' , True}
print(a)          # {'Hyd', True, 25, 10.8}
a.remove('Hyd')

# discard()  method  demo  program (Home  work)
a = {25 , 10.8 , 'Hyd' , True}
print(a)          # {'Hyd', True, 25, 10.8}
a.discard('Hyd')
print(a)          # {True, 25, 10.8}
a.discard('Sec')
print(a)          #  unchanged
a.remove('Sec')   # Error
